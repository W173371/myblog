# ---------------------------------------------------------------------------
#  一、基本信息
# ---------------------------------------------------------------------------

# 文章标题（必填），显示在页面顶部和浏览器标签页
title = "{{ replace .File.ContentBaseName "-" " " | title }}"

# 副标题（可选），显示在标题下方，灰色小字
subtitle = ""

# 发布日期（必填），格式：YYYY-MM-DD 或 ISO 8601
date = "{{ .Date }}"

# 最后修改日期（可选），留空则自动从 Git 或文件系统获取
# 格式同 date，例如 "2026-05-27" 或 "2026-05-27T14:30:00+08:00"
lastmod = ""

# 文章 Slug（可选），自定义 URL 中的文章名
# 留空则自动使用文件名，例如 "my-first-post"
slug = ""

# 自定义 URL 路径（可选），会完全替换默认的文章路径
# 例如 "/p/special" 会让文章直接访问 /p/special/
url = ""

# 文章权重（可选），数值越大排序越靠前，默认 0
# 用于自定义文章在列表中的排列顺序
weight = 0

# 侧边栏目录的排序权重（可选），设为 0 时按默认顺序
# 仅影响 sidebar 中的目录排序
linkTitle = ""


# ---------------------------------------------------------------------------
#  二、状态与可见性
# ---------------------------------------------------------------------------

# 是否草稿（true=不发布，仅 hugo server -D 可见 / false=公开发布）
draft = true

# 是否在首页隐藏这篇文章（true=隐藏 / false=正常显示）
hiddenFromHomePage = false

# 是否在搜索结果中隐藏（true=隐藏 / false=正常）
hiddenFromSearch = false

# 是否在"相关文章"推荐中隐藏（true=隐藏 / false=正常）
hiddenFromRelated = false

# 是否在 RSS/Atom/JSON Feed 中隐藏（true=隐藏 / false=正常）
hiddenFromFeed = false


# ---------------------------------------------------------------------------
#  三、作者信息（覆盖全局设置）
# ---------------------------------------------------------------------------

[author]
  # 作者名称，留空则使用 hugo.toml 中的 params.author.name
  name = ""
  # 作者主页链接
  link = ""
  # 作者邮箱（可用于 Gravatar 头像）
  email = ""
  # 作者头像 URL，留空则使用全局设置
  avatar = ""


# ---------------------------------------------------------------------------
#  四、SEO 与描述
# ---------------------------------------------------------------------------

# 文章描述（可选），用于 SEO meta description 和文章摘要
# 留空则自动截取正文前 70 字
description = ""

# 文章关键词（可选），用于 SEO meta keywords
# 格式：["关键词1", "关键词2", "关键词3"]
keywords = []

# 自定义文章摘要（可选），会替代自动提取的摘要
# 也可以在正文中插入 <!--more--> 来手动截断摘要
summary = ""


# ---------------------------------------------------------------------------
#  五、分类与标签
# ---------------------------------------------------------------------------

# 文章分类（必填），可选：数字生活 / 梅花易数 / 杂文
# 一篇文章可以属于多个分类
categories = []

# 文章标签（可选），例如 ["Hugo", "博客", "建站"]
tags = []

# 文章合集/系列（可选），用于文章连载
# 同一合集的文章会自动显示"上一篇/下一篇"导航
collections = []


# ---------------------------------------------------------------------------
#  六、封面图与预览图
# ---------------------------------------------------------------------------

# 列表页预览图（可选），显示在文章卡片上
# 图片放在 static/images/ 或文章同级目录
# 例如 "/images/cover.jpg" 或 "cover.jpg"
featuredImagePreview = ""

# 文章顶部封面大图（可选），显示在文章标题上方
# 留空则使用 featuredImagePreview 作为封面图
featuredImage = ""

# 文章配图列表（可选），用于 SEO Open Graph 分享预览
# 格式：["/images/og-1.jpg", "/images/og-2.jpg"]
images = []


# ---------------------------------------------------------------------------
#  七、评论、分享、赞赏
# ---------------------------------------------------------------------------

# 是否允许评论（可选），留空使用全局设置
# true=强制开启 / false=强制关闭 / 留空=跟随全局
comment = true

# 是否显示分享按钮，留空跟随全局设置
share = true

# 是否显示打赏，留空跟随全局设置
reward = true


# ---------------------------------------------------------------------------
#  八、文章功能开关
# ---------------------------------------------------------------------------

# 是否显示目录（TOC），留空跟随全局
# true=强制显示 / false=强制隐藏
toc = true

# 是否启用数学公式渲染（KaTeX / MathJax），留空跟随全局
math = true

# 是否启用图片灯箱（lightgallery），留空跟随全局
# true=只有带 title 的图片可放大 / "force"=所有图片可放大 / false=禁用
lightgallery = true

# 是否显示字数统计，留空跟随全局
wordCount = true

# 是否显示阅读时间，留空跟随全局
readingTime = true

# 文章末尾结束标记，留空跟随全局
# 设为 "" 表示不显示结束标记
endFlag = ""

# 是否显示作者头像，留空跟随全局
authorAvatar = true

# 是否使用 Twemoji 替换系统 Emoji
twemoji = false

# 是否启用注音扩展语法（ruby 标签）
ruby = true

# 是否启用分数扩展语法（如 1/2 → ½）
fraction = true

# 是否启用 FontAwesome 图标语法
fontawesome = true


# ---------------------------------------------------------------------------
#  九、加密保护
# ---------------------------------------------------------------------------

# 访问密码（可选），设置后需要输入密码才能查看文章内容
# 密码使用 SHA256 加密存储
# 示例：password = "mypassword"
password = ""

# 密码提示信息（可选），显示在密码输入框上方
# 支持 Markdown 和 HTML
message = ""

#  十、转载声明
# ---------------------------------------------------------------------------

[repost]
  # 是否启用转载声明（显示"本文转载自 xxx"）
  enable = false
  # 原文链接
  url = ""

#  十一、资源引用（高级用法）
# ---------------------------------------------------------------------------

  [page.library.js]
    # 自定义 JS 文件
    # myjs = "myjs.js"
